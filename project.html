项目1：简易计算器

涉及知识点：函数、条件判断、循环、输入输出

def calculator():
    print("=== 简易计算器 ===")
    print("支持的运算：+, -, *, /")

    while True:
        try:
            a = float(input("请输入第一个数字："))
            op = input("请输入运算符（输入 q 退出）：")
            if op == "q":
                print("退出计算器")
                break
            b = float(input("请输入第二个数字："))

            if op == "+":
                result = a + b
            elif op == "-":
                result = a - b
            elif op == "*":
                result = a * b
            elif op == "/":
                if b == 0:
                    print("错误：除数不能为0")
                    continue
                result = a / b
            else:
                print("不支持的运算符")
                continue

            print(f"{a} {op} {b} = {result}")
            print("-" * 20)

        except ValueError:
            print("错误：请输入有效的数字")

if __name__ == "__main__":
    calculator()

项目2：猜数字游戏

涉及知识点：random 库、while 循环、条件判断、异常处理

import random

def guess_number():
    target = random.randint(1, 100)
    attempts = 0
    print("=== 猜数字游戏 ===")
    print("我已经想好了一个1-100之间的数字")

    while True:
        try:
            guess = int(input("请猜一个数字："))
            attempts += 1

            if guess < 1 or guess > 100:
                print("请输入1-100之间的数字")
                continue

            if guess < target:
                print("小了，再大一点")
            elif guess > target:
                print("大了，再小一点")
            else:
                print(f"恭喜你猜中了！答案是{target}")
                print(f"你一共猜了{attempts}次")
                break

        except ValueError:
            print("请输入有效的数字")

if __name__ == "__main__":
    guess_number()

项目3：记事本读写工具

涉及知识点：文件操作、with 语句、try-except

import os

NOTE_FILE = "notes.txt"

def write_note():
    content = input("请输入笔记内容：")
    with open(NOTE_FILE, "a", encoding="utf-8") as f:
        f.write(content + "\n")
    print("笔记已保存")

def read_notes():
    if not os.path.exists(NOTE_FILE):
        print("暂无笔记")
        return
    with open(NOTE_FILE, "r", encoding="utf-8") as f:
        notes = f.read()
        if notes:
            print("=== 我的笔记 ===")
            print(notes)
        else:
            print("暂无笔记")

def delete_notes():
    if os.path.exists(NOTE_FILE):
        os.remove(NOTE_FILE)
        print("所有笔记已删除")
    else:
        print("暂无笔记可删除")

def main():
    while True:
        print("\n=== 记事本工具 ===")
        print("1. 写笔记")
        print("2. 查看笔记")
        print("3. 删除所有笔记")
        print("4. 退出")
        choice = input("请选择：")

        if choice == "1":
            write_note()
        elif choice == "2":
            read_notes()
        elif choice == "3":
            delete_notes()
        elif choice == "4":
            print("再见！")
            break
        else:
            print("无效选择")

if __name__ == "__main__":
    main()

项目4：批量重命名文件

涉及知识点：os 库、列表操作、字符串处理

import os

def batch_rename():
    folder = input("请输入文件夹路径（留空为当前目录）：").strip()
    if not folder:
        folder = "."

    if not os.path.exists(folder):
        print("文件夹不存在")
        return

    prefix = input("请输入文件名前缀：")
    ext = input("请输入要修改的文件扩展名（如 .txt）：")
    start_num = int(input("请输入起始编号（默认1）：") or "1")

    files = [f for f in os.listdir(folder)
             if f.endswith(ext) and os.path.isfile(os.path.join(folder, f))]
    files.sort()

    if not files:
        print(f"没有找到 {ext} 文件")
        return

    print(f"找到 {len(files)} 个文件")
    for i, old_name in enumerate(files, start=start_num):
        new_name = f"{prefix}_{i:03d}{ext}"
        old_path = os.path.join(folder, old_name)
        new_path = os.path.join(folder, new_name)
        os.rename(old_path, new_path)
        print(f"重命名：{old_name} → {new_name}")

    print("全部重命名完成")

if __name__ == "__main__":
    batch_rename()

项目5：学生信息管理系统

涉及知识点：OOP、文件存储、函数模块化

import json
import os

DATA_FILE = "students.json"

class Student:
    def __init__(self, sid, name, age, grade):
        self.sid = sid
        self.name = name
        self.age = age
        self.grade = grade

    def to_dict(self):
        return {"id": self.sid, "name": self.name,
                "age": self.age, "grade": self.grade}

class StudentManager:
    def __init__(self):
        self.students = []
        self.load_data()

    def load_data(self):
        if os.path.exists(DATA_FILE):
            with open(DATA_FILE, "r", encoding="utf-8") as f:
                data = json.load(f)
                self.students = [Student(**s) for s in data]

    def save_data(self):
        with open(DATA_FILE, "w", encoding="utf-8") as f:
            json.dump([s.to_dict() for s in self.students],
                      f, ensure_ascii=False, indent=2)

    def add_student(self):
        sid = input("学号：")
        name = input("姓名：")
        age = int(input("年龄："))
        grade = float(input("成绩："))
        self.students.append(Student(sid, name, age, grade))
        self.save_data()
        print("添加成功")

    def show_all(self):
        if not self.students:
            print("暂无学生信息")
            return
        print(f"{'学号':<8}{'姓名':<8}{'年龄':<6}{'成绩':<6}")
        print("-"*30)
        for s in self.students:
            print(f"{s.sid:<8}{s.name:<8}{s.age:<6}{s.grade:<6}")

    def search(self):
        keyword = input("请输入学号或姓名：")
        results = [s for s in self.students
                   if keyword in s.sid or keyword in s.name]
        if results:
            for s in results:
                print(f"{s.sid} {s.name} {s.age}岁 {s.grade}分")
        else:
            print("未找到")

    def delete_student(self):
        sid = input("请输入要删除的学号：")
        for s in self.students:
            if s.sid == sid:
                self.students.remove(s)
                self.save_data()
                print("已删除")
                return
        print("未找到该学号")

    def menu(self):
        while True:
            print("\n=== 学生信息管理系统 ===")
            print("1. 添加学生")
            print("2. 查看所有学生")
            print("3. 搜索学生")
            print("4. 删除学生")
            print("5. 退出")
            choice = input("请选择：")
            if choice == "1": self.add_student()
            elif choice == "2": self.show_all()
            elif choice == "3": self.search()
            elif choice == "4": self.delete_student()
            elif choice == "5":
                print("再见！")
                break
            else:
                print("无效选择")

if __name__ == "__main__":
    manager = StudentManager()
    manager.menu()

项目6：天气查询工具

涉及知识点：requests 爬虫、JSON 解析、API 调用

import requests

def get_weather(city):
    # 使用 wttr.in 免费天气 API（无需 key）
    url = f"https://wttr.in/{city}?format=%C+%t+%w+%h"

    try:
        resp = requests.get(url, timeout=10)
        if resp.status_code == 200:
            return resp.text.strip()
        else:
            return "查询失败"
    except Exception as e:
        return f"错误：{e}"

def main():
    print("=== 天气查询工具 ===")
    print("支持中文城市名，如：北京、上海、深圳")

    while True:
        city = input("\n请输入城市名（输入 q 退出）：").strip()
        if city.lower() == "q":
            print("再见！")
            break
        if not city:
            continue

        weather = get_weather(city)
        print(f"{city}天气：{weather}")

if __name__ == "__main__":
    main()

项目7：Excel 数据统计工具

涉及知识点：openpyxl、数据统计、表格处理

from openpyxl import Workbook, load_workbook
import os

def create_sample_data():
    wb = Workbook()
    ws = wb.active
    ws.title = "销售数据"
    ws.append(["月份", "产品", "销量", "金额"])
    ws.append(["1月", "A", 120, 12000])
    ws.append(["1月", "B", 85, 8500])
    ws.append(["2月", "A", 150, 15000])
    ws.append(["2月", "B", 95, 9500])
    ws.append(["3月", "A", 110, 11000])
    ws.append(["3月", "B", 130, 13000])
    wb.save("销售数据.xlsx")
    return "销售数据.xlsx"

def analyze_excel(filename):
    if not os.path.exists(filename):
        return "文件不存在"

    wb = load_workbook(filename, data_only=True)
    ws = wb.active

    data = []
    for row in ws.iter_rows(min_row=2, values_only=True):
        if row[0] is not None:
            data.append(row)

    if not data:
        return "无数据"

    total_sales = sum(row[2] for row in data)
    total_amount = sum(row[3] for row in data)
    avg_amount = total_amount / len(data) if data else 0

    products = {}
    for row in data:
        product = row[1]
        if product not in products:
            products[product] = {"销量": 0, "金额": 0}
        products[product]["销量"] += row[2]
        products[product]["金额"] += row[3]

    result = f"总销量：{total_sales}\n"
    result += f"总金额：{total_amount}\n"
    result += f"平均金额：{avg_amount:.2f}\n"
    result += "各产品销售统计：\n"
    for p, s in products.items():
        result += f"  {p}：销量{s['销量']}，金额{s['金额']}\n"

    return result

if __name__ == "__main__":
    file = create_sample_data()
    print(f"已创建示例文件：{file}")
    print("\n统计分析结果：")
    print(analyze_excel(file))

项目8：Flask 个人博客（后端）

涉及知识点：Flask Web 框架、路由、模板渲染、请求处理

from flask import Flask, render_template, request, redirect

app = Flask(__name__)

# 模拟数据库
posts = []
post_id = 1

@app.route("/")
def home():
    return """
    <h1>个人博客</h1>
    <a href="/posts">查看所有文章</a> |
    <a href="/new">写文章</a>
    """

@app.route("/posts")
def list_posts():
    html = "<h1>文章列表</h1><ul>"
    for p in posts:
        html += f'<li><a href="/post/{p["id"]}">{p["title"]}</a></li>'
    html += "</ul><a href='/new'>写新文章</a>"
    return html

@app.route("/post/<int:id>")
def show_post(id):
    for p in posts:
        if p["id"] == id:
            return f"<h1>{p['title']}</h1><p>{p['content']}</p>"
    return "<h1>404 - 文章不存在</h1>", 404

@app.route("/new", methods=["GET", "POST"])
def new_post():
    global post_id
    if request.method == "POST":
        title = request.form["title"]
        content = request.form["content"]
        posts.append({"id": post_id, "title": title, "content": content})
        post_id += 1
        return redirect("/posts")
    return """
    <h1>写新文章</h1>
    <form method="post">
        <p>标题：<input type="text" name="title" required></p>
        <p>内容：</p>
        <textarea name="content" rows="10" cols="50" required></textarea>
        <p><button type="submit">发布</button></p>
    </form>
    <a href="/posts">返回文章列表</a>
    """

if __name__ == "__main__":
    app.run(debug=True)

项目9：简易图片爬虫

涉及知识点：requests、BeautifulSoup、文件下载、异常处理

import requests
from bs4 import BeautifulSoup
import os
from urllib.parse import urljoin

def download_images(url, save_folder="images"):
    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
    }

    try:
        resp = requests.get(url, headers=headers, timeout=10)
        resp.encoding = "utf-8"
        soup = BeautifulSoup(resp.text, "html.parser")
    except Exception as e:
        print(f"请求失败：{e}")
        return

    os.makedirs(save_folder, exist_ok=True)

    img_tags = soup.find_all("img")
    downloaded = 0

    for i, img in enumerate(img_tags):
        src = img.get("src")
        if not src:
            continue
        img_url = urljoin(url, src)

        if not img_url.startswith("http"):
            continue

        try:
            img_resp = requests.get(img_url, headers=headers, timeout=10)
            if img_resp.status_code == 200:
                ext = os.path.splitext(img_url)[1]
                if ext.lower() not in [".jpg", ".jpeg", ".png", ".gif", ".webp"]:
                    ext = ".jpg"
                filename = os.path.join(save_folder, f"img_{i}{ext}")
                with open(filename, "wb") as f:
                    f.write(img_resp.content)
                downloaded += 1
                print(f"已下载：{filename}")
        except Exception as e:
            print(f"下载失败：{img_url} - {e}")

    print(f"\n共下载 {downloaded} 张图片到 {save_folder} 文件夹")

if __name__ == "__main__":
    url = input("请输入网页URL：")
    download_images(url)

项目10：自动化签到脚本

涉及知识点：requests 会话、定时任务、文件存储状态

"""
模拟签到脚本 - 自动记录签到状态
实际使用时替换为对应网站的 API
"""
import json
import os
from datetime import datetime

RECORD_FILE = "checkin_record.json"

class AutoCheckin:
    def __init__(self):
        self.records = self.load_records()

    def load_records(self):
        if os.path.exists(RECORD_FILE):
            with open(RECORD_FILE, "r", encoding="utf-8") as f:
                return json.load(f)
        return {"total_days": 0, "last_checkin": None, "history": []}

    def save_records(self):
        with open(RECORD_FILE, "w", encoding="utf-8") as f:
            json.dump(self.records, f, ensure_ascii=False, indent=2)

    def checkin(self, username):
        today = datetime.now().strftime("%Y-%m-%d")

        # 检查是否已签到
        if self.records.get("last_checkin") == today:
            return f"{username}，今天已经签到过了"

        # 模拟签到请求
        print(f"正在为 {username} 执行签到...")
        # TODO: 替换为真实的 API 请求
        # resp = requests.post("https://example.com/checkin", data={"user": username})

        self.records["total_days"] += 1
        self.records["last_checkin"] = today
        self.records["history"].append({
            "date": today,
            "time": datetime.now().strftime("%H:%M:%S"),
            "username": username
        })
        self.save_records()

        return f"{username} 签到成功！连续签到 {self.records['total_days']} 天"

    def show_status(self):
        print("\n=== 签到状态 ===")
        print(f"总签到天数：{self.records['total_days']}")
        print(f"上次签到：{self.records['last_checkin'] or '从未签到'}")
        if self.records["history"]:
            print("最近记录：")
            for h in self.records["history"][-5:]:
                print(f"  {h['date']} {h['time']} - {h['username']}")

def main():
    checker = AutoCheckin()

    while True:
        print("\n=== 自动化签到系统 ===")
        print("1. 执行签到")
        print("2. 查看签到状态")
        print("3. 退出")
        choice = input("请选择：")

        if choice == "1":
            username = input("请输入用户名：")
            result = checker.checkin(username)
            print(result)
        elif choice == "2":
            checker.show_status()
        elif choice == "3":
            print("再见！")
            break
        else:
            print("无效选择")

if __name__ == "__main__":
    main()
