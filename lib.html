数据分析方向

pip install numpy NumPy - 科学计算基础
import numpy as np

# 创建数组
arr = np.array([1, 2, 3, 4, 5])
print(arr)           # [1 2 3 4 5]
print(arr.shape)     # (5,)

# 二维数组
matrix = np.array([[1, 2], [3, 4]])
print(matrix.shape)  # (2, 2)

# 常用函数
zeros = np.zeros((3, 3))  # 全0矩阵
ones = np.ones((2, 4))    # 全1矩阵
nums = np.arange(10)      # [0 1 2 ... 9]
random = np.random.randn(5)  # 正态分布随机数

# 数组运算
a = np.array([1, 2, 3])
b = np.array([4, 5, 6])
print(a + b)   # [5 7 9]
print(a * b)   # [4 10 18]
print(a.mean())  # 2.0
print(a.sum())   # 6

import pandas as pd

# Series（一维）
s = pd.Series([10, 20, 30, 40], index=["a", "b", "c", "d"])

print(df["年龄"].mean()) # 年龄平均值

# 筛选
print(df[df["成绩"] > 90])  # 成绩 > 90 的行

网络爬虫方向

pip install requests Requests - HTTP 请求库
import requests

# GET 请求
resp = requests.get("https://api.github.com")
print(resp.status_code)   # 200
print(resp.json())        # JSON 格式响应

# POST 请求
data = {"username": "admin", "password": "123456"}
resp = requests.post("https://httpbin.org/post", data=data)
print(resp.text)

# 请求头
headers = {"User-Agent": "Mozilla/5.0"}
resp = requests.get("https://example.com", headers=headers)

# 超时设置
resp = requests.get("https://example.com", timeout=5)

自动化方向

pip install openpyxl Openpyxl - Excel 操作
from openpyxl import Workbook, load_workbook

# 创建 Excel 文件
wb = Workbook()
ws = wb.active
ws.title = "成绩表"

# 写入数据
ws["A1"] = "姓名"
ws["B1"] = "成绩"
ws.append(["小明", 95])
ws.append(["小红", 88])
ws.append(["小刚", 92])

# 保存
wb.save("成绩表.xlsx")

# 读取 Excel
wb2 = load_workbook("成绩表.xlsx")
ws2 = wb2.active
for row in ws2.iter_rows(values_only=True):
    print(row)  # ('姓名', '成绩'), ('小明', 95), ...

Web 开发方向

pip install flask Flask - 轻量 Web 框架
from flask import Flask, render_template, request

app = Flask(__name__)

@app.route("/")
def home():
    return "<h1>Hello Flask!</h1>"

@app.route("/user/<name>")
def user(name):
    return f"<h1>你好，{name}！</h1>"

@app.route("/login", methods=["GET", "POST"])
def login():
    if request.method == "POST":
        username = request.form["username"]
        return f"登录成功：{username}"
    return """
        <form method="post">
            <input type="text" name="username">
            <button type="submit">登录</button>
        </form>
    """

if __name__ == "__main__":
    app.run(debug=True)
